!Configure login access for switch via CLI, console and SSH.
hostname DistributionSwitch1
no ip domain-lookup
enable secret Test1234
line console 0
  password Test1234
  login
  exit
line vty 0 4
  password Test1234
  login
  transport input ssh
  exit

!Enable IP address routing on switch
ip routing

!Configure interface fa0/1 connected to Core Router
interface g0/1
  description Uplink to Core Router g0/1
  no switchport
  ip address 10.0.0.1 255.255.255.252
  no shutdown
  exit

!Configure interface fa0/1 connected to Access Switch 1-1
interface g0/2
  description Trunk to AccessSwitch1-1
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk allowed vlan 10,20
  no shutdown
  exit

!Configure interface fa0/1 connected to Access Switch 1-2
interface fa0/1
  description Trunk to AccessSwitch1-2
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk allowed vlan 30,40
  no shutdown
  exit

!Configure and close unused ports
interface range fa0/2-24
  switchport access vlan 1
  shutdown
  exit

!Configure all VLANs
vlan 10
  name Staff
  exit
vlan 20
  name VoIP
  exit
vlan 30
  name Servers
  exit
vlan 40
  name Wireless
  exit

!Configure SVI and default gateway for each VLAN
interface vlan 10
  ip address 10.0.10.1 255.255.255.0
  no shutdown
  exit
interface vlan 20
  ip address 10.0.20.1 255.255.255.0
  no shutdown
  exit
interface vlan 30
  ip address 10.0.30.1 255.255.255.0
  no shutdown
  exit
interface vlan 40
  ip address 10.0.40.1 255.255.255.0
  no shutdown
  exit

!Configure OSPF
router ospf 1
  router-id 2.2.2.2
  network 10.0.0.0 0.0.0.3 area 0
!Advertise VLAN subnets
  network 10.0.10.0 0.0.0.255 area 0
  network 10.0.20.0 0.0.0.255 area 0
  network 10.0.30.0 0.0.0.255 area 0
  network 10.0.40.0 0.0.0.255 area 0
!Configure passive VLAN interface to prevent OSPF flooding downstream
  passive-interface vlan 10
  passive-interface vlan 20
  passive-interface vlan 30
  passive-interface vlan 40
  exit
end

!Save and copy running config
copy running-config startup-config
startup-config
