hostname AccessSwitch1-1
no ip domain-lookup
spanning-tree portfast bpduguard default

!Configure VLANs
vlan 10
  name Staff
  exit
vlan 20
  name VoIP
  exit

!Configure uplink interface g0/1 connected to Distribution Switch 1
interface g0/1
  description Uplink to Distribution Switch 1
  switchport mode trunk
  switchport trunk allowed vlan 10,20
  no shutdown
  exit

!Configure interfaces for VLAN 10 
interface range fa0/1-10
  switchport mode access
  switchport access vlan 10
  no shutdown
  exit

!Configure interfaces for VLAN 20
interface range fa0/11-15
  switchport mode access
  switchport access vlan 20
  no shutdown
  exit

!Disable unused ports
interface range fa0/16-24
  switchport access vlan 1
  shutdown
  exit
end

!Save and copy running config
copy running-config startup-config
startup-config
